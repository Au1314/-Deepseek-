## 三、API 请求与响应结构

### 3.1 请求端

所有三个模块统一调用 `fetchSQLWorkflow(intention, userId)`：

```javascript
// api.js:58-59
async function fetchSQLWorkflow(intention, userId) {
    return await fetchWorkflowData({ intention: intention }, userId, SQL_AUTH_TOKEN);
}

// ↓ 内部调用 api.js:11-51
POST http://YOUR_DIFY_HOST/v1/workflows/run

Headers: {
    Authorization: "Bearer app-YOUR_SQL_TOKEN",
    Content-Type: "application/json"
}

Body: {
    "inputs": {
        "intention": "获取医生数据"       // 每个模块传不同的意图字符串
    },
    "response_mode": "blocking",         // 阻塞等待
    "user": "current_user"
}
```

### 3.2 响应端

```javascript
// api.js:34-45 — 统一解析层
const result = await response.text();
const responseData = JSON.parse(result);

// Dify Workflow 原始返回结构
{
    "data": {
        "outputs": {
            "body": "{\"result\": [...]}"   // 可能是 JSON 字符串
            // 或
            "body": { "result": [...] }      // 可能是已解析的对象
        }
    }
}

// 解析逻辑
if (typeof responseData.data.outputs.body === 'string') {
    return JSON.parse(responseData.data.outputs.body);  // 字符串→对象
}
return responseData.data.outputs.body;                   // 直接返回对象

// 最终返回给 main.js 的数据格式
{
    "result": [
        { ... },   // 每条记录一个对象
        { ... },
        ...
    ]
}
```

### 3.3 三个模块的意图与返回字段对比

| 模块 | intention | 返回数组 result[] | 每条字段 |
|------|-----------|-----------------|---------|
| 医师团队 | `"获取医生数据"` | 不限条数 | `image_url`, `doctor_name`, `title`, `department`, `chat_token`, `info_id` |
| 健康科普 | `"查询3篇文章信息"` | 固定3条 | `article_id`, `cover_url`, `title`, `content`, `views` |
| 糖尿病类型 | `"获取糖尿病类型"` | 不限条数 | `type_id`, `type_name`, `img`, `pathogenesis` |

## 五、缓存机制

```javascript
// main.js:3-22
function getCachedData(key) {
    const cached = sessionStorage.getItem(key);
    if (!cached) return null;
    const { data, timestamp } = JSON.parse(cached);
    if (Date.now() - timestamp > 3600000) {    // 1小时过期
        sessionStorage.removeItem(key);
        return null;
    }
    return data;
}

function setCachedData(key, data) {
    sessionStorage.setItem(key, JSON.stringify({
        data,                                   // 实际数据
        timestamp: Date.now()                   // 缓存时间戳
    }));
}
```

| 模块 | 缓存 Key | 有效期 |
|------|---------|--------|
| 医师团队 | `doctors_data` | 1小时 |
| 健康科普 | `articles_data` | 1小时 |
| 糖尿病类型 | `diabetes_types_data` | 1小时 |

## 六、图片加载与容错

```javascript
// 统一图片处理函数
function normalizeImageUrl(url, defaultImg) {
    if (!url || typeof url !== 'string' || url.trim() === '' || url === 'null' || url === 'undefined') {
        return defaultImg;                    // 过滤各种异常值
    }
    if (url.startsWith('http://') || url.startsWith('https://') || url.startsWith('/')) {
        return url;                           // 完整URL或绝对路径直接使用
    }
    return '/' + url.replace(/^\.?\//, '');   // 相对路径补全为绝对路径
}
```

**双层容错**：
1. JS 层 `normalizeImageUrl()` 过滤异常 URL 值
2. HTML 层 `onerror` 事件：图片加载失败时替换为默认图

## 八、常见问题排查清单

| 现象 | 可能原因 | 检查位置 |
|------|---------|---------|
| 模块显示"加载中..."不消失 | API 请求失败或跨域 | api.js 第 18-27 行的 fetch 调用 |
| 图片裂开不显示 | `cover_url` 为字符串 `"null"` | main.js → `normalizeImageUrl()` |
| 文章列表整个空白 | `article.content` 为 null 导致 `.substring()` 报错 | main.js 第 88 行的安全判断 |
| 数据不是最新的 | sessionStorage 缓存 1 小时未过期 | main.js 第 9 行的过期判断 |
| 轮播图被遮挡/不显示 | `.swiper` 未设置 height | main.css 第 72-75 行 |
| "查看全部/更多/全部" 链接无效 | 三个 `href="#"` 无 JS 绑定 | main.html 第 60/72/84 行 |
